function inspect(path, max, expected, analyze = () => ({})) {
  const named = fs.lstatSync(path,{bigint:true})
  const fd = fs.openSync(path,fs.constants.O_RDONLY|fs.constants.O_NOFOLLOW|fs.constants.O_NONBLOCK)
  try {
    const before = fs.fstatSync(fd,{bigint:true})
    assert(before.isFile() && named.isFile()); assert.equal(before.uid,BigInt(process.getuid())); assert.equal(before.nlink,1n); assert.equal(before.mode & 0o7777n,0o600n)
    assert.deepEqual(metadata(before),metadata(named)); assert(before.size >= 0n && before.size <= BigInt(max))
    if (expected?.metadata) for (const [key,value] of Object.entries(expected.metadata)) assert.equal(String(before[key]),value,key)
    const data = Buffer.alloc(Number(before.size)+1); let used=0
    while (used<data.length) { const n=fs.readSync(fd,data,used,data.length-used,used); if(n===0)break; assert(n>0); used+=n }
    assert.equal(used,Number(before.size))
    const bytes=data.subarray(0,used), sha256=digest(bytes)
    if(expected?.sha256)assert.equal(sha256,expected.sha256)
    const result=analyze(bytes)
    assert.deepEqual(metadata(fs.fstatSync(fd,{bigint:true})),metadata(before))
    const after=fs.lstatSync(path,{bigint:true}); assert(after.isFile()); assert.deepEqual(metadata(after),metadata(before))
    return {path,...metadata(before),sha256,...result}
  } finally {fs.closeSync(fd)}
}
